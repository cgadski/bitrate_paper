\section{Proof of \Cref{prop:necessary-cond} \label{appendix:necessary}}

In our discussion of Gaussian channels in \Cref{sec:information}, we recalled that when an individual coordinate $X$ of unit variance suffers interference of power $P,$ it can convey at most $\ln(1 + P^{-1})/2$ nats to a receiver. In terms of mutual information, this means that
$$
I(X; X + Z) \le \frac 1 2 \ln (1 + P^{-1}) = \frac{1}{2 P} + O(P^{-2})
$$
for large $P.$ We begin by establishing a similar bound on mutual information between a superposition of Rademacher codewords and one of its summands.

\begin{proposition} \label{prop:mutual}
Let $F_1, \dots, F_k$ be $k$ independent Rademacher codewords, as in the definition of a Rademacher dictionary. Then for any random variable $Y$ independent from $(F_1, \dots, F_k)$ and supported on the set $\{-1/\sqrt{d}, 1/\sqrt{d}\}^d,$
$$
I\left(Y; Y + \sum_{i = 1}^k F_i \right) \le d \left(\frac 1 {2 k} + O(k^{-2})\right),
$$
where the remainder of $O(k^{-2})$ depends only on $k.$
\end{proposition}

\begin{proof}
Let $Z = F_1 + \dots + F_k.$ By independence,
$$
I(Y; Y + Z) = H(Y + Z) - H(Y + Z | Y) = H(Y + Z) - H(Z).
$$
Now, let $\pi_i$ be the projection onto the $i$th coordinate. Since the coordinates of $Z$ are independent,
$$
H(Z) = \sum_{i = 1}^d H(\pi_i(Z)),
$$
Furthermore, by subadditivity,
$$
H(Y + Z) \le \sum_{i = 1}^d H(\pi_i(Y + Z)).
$$
The entropy of $\pi_i(Y + Z) = \pi_i(Y) + \pi_i(Z)$ is unaffected by applying a rescaling where $\pi_i(Y) \in \{0, 1\}$ and $\pi_i(Z)$ has a binomial distribution with $k$ trials. Overall, it is enough to prove that
$$
I(X; B_k + X) = H(B_k + X) - H(B_k) \le \frac{1}{2 k} + O(k^{-2})
$$
where $B_k$ has a Binomial distribution with $k$ trials and $X$ is independent from $B_k$ and supported on $\{0, 1\}.$

It is simple to check that $H(B_k + X)$ is maximized when $X$ is uniformly distributed over $\{0, 1\}.$ Let $X$ take this distribution. Then
$$
\Pp(X = 1|B_k + X) = \frac{B_k + X}{k + 1}.
$$
So, where $h(p) = -p \ln p - (1 - p) \ln (1 - p),$ we have the explicit expression
$$
I(X; B_k + X) = H(X) - H(X|B_k + X) = \ln 2 - \E \left[ h\left( \frac{B_k + X}{k + 1}\right)\right].
$$

Now, the power series for binary entropy gives, for $|t| \le 1/2,$
$$
h(1/2 + t) = \ln 2 - \sum_{m = 1}^\infty \frac{(2t)^{2m}}{2m(2m - 1)}.
$$
Substituting $t = \frac{B_k + X}{k + 1} - \frac{1}{2}$ and taking expectations,
$$
I(X; B_k + X) = \sum_{m = 1}^\infty \frac{\mu_{2m}}{2m(2m - 1)}, \qquad \mu_{2m} = \E\left[\left(\frac{2(B_k + X)  - (k + 1)}{k + 1}\right)^{2m}\right].
$$
Since $\left|\frac{2(B_k + X) - (k + 1)}{k + 1}\right| \le 1,$ we have $\mu_{2m} \le \mu_4$ for all $m \ge 2.$ The leading term is
$$
\mu_2 = \frac{4 \operatorname{Var}(B_{k + 1})}{(k + 1)^2} = \frac{1}{k + 1},
$$
and the fourth moment satisfies $\mu_4 = (3k + 1)/(k + 1)^3 = O(k^{-2}).$ Therefore,
$$
I(X; B_k + X) = \frac{1}{2(k + 1)} + O(k^{-2}) = \frac{1}{2k} + O(k^{-2}).
$$
\end{proof}

Now we proceed to the proof of \Cref{prop:necessary-cond}.

\color{blue} \begin{proposition*}
    Let $C > 0$ be arbitrary. Over any regime where
    $$
    d \le C k \ln N, \quad \omega(1) \le k < N/2,
    $$
    for Rademacher dictionaries it holds that
    $$
    \liminf_{N \to \infty} b(d, k, N; \text{top-$k$}) \ge 1 - \frac C 2.
    $$
\end{proposition*}

\color{blue} \begin{proof}
It suffices to show that for all integers $d, k, N \ge 1,$
$$
b(d, k + 1, N + k; \text{top-}k) \ge 1 - \frac{\ln 2}{\ln N} - \frac{d}{2 k \ln N}(1 + O(k^{-1})).
$$
Indeed, substituting $k' = k - 1$ and $N' = N - k + 1$ (so that $k' + 1 = k$ and $N' + k' = N$), when $k \le N/2$ this gives
$$
b(d, k, N; \text{top-}k) \ge 1 - \frac{\ln 2}{\ln (N - k + 1)} - \frac{d}{2 (k - 1) \ln (N - k + 1)}(1 + O(k^{-1})).
$$
Under the regime $d \le C k \ln N$ and $\omega(1) \le k \le N/2,$ we conclude overall that
$$
b(d, k, N, \text{top-}k) \ge 1 - o(1) - \frac{C}{2}.
$$

We now prove the claim. Let $F \in \R^{d \times (N + k)}$ be a Rademacher dictionary, and let $(F^0, F^1)$ denote the sub-matrices of $F$ containing the first $N$ and the last $k$ columns respectively. Let $K \in [N]$ be uniformly random, independent from $F,$ and let $X \in \R^{N + k}$ have coordinates
$$
X_i = \begin{cases}
1 : i = K \lor i > N \\
0 : \text{otherwise}.
\end{cases}
$$
Denote $(X^0, X^1)$ for its first $N$ and last $k$ coordinates of $X,$ so that $F X = F^0 X^0 + F^1 X^1.$ Finally, define a function $G \colon \R^{d} \to [N]$ by
$$
G(y) = \arg \max_{i \in [N]} \langle y, F_i \rangle,
$$
breaking ties arbitrarily.

If top-$k$ decoding succeeds at recovering $X$ from $F X,$ then in particular $G(FX) = K.$ It follows that
$$
b(d, k + 1, N + k; \text{top-}k) \ge \Pp(G(FX) \neq K).
$$
However, by Fano's inequality,
$$
\Pp(G(FX) \neq K) \ge 1 - \frac{\ln 2 + I(F^0 X^0; F X)}{\ln N}.
$$
Now, $F^1 X^1$ is a sum of $k$ independent Rademacher vectors that is independent from $F^0 X^0 \in \{-1/\sqrt{d}, 1/\sqrt{d}\}^d.$ Applying \Cref{prop:mutual} to bound $I(F^0 X^0; FX) \le d(1/(2k) + O(k^{-2}))$ yields
$$
b(d, k + 1, N + k; \text{top-}k) \ge 1 - \frac{\ln 2}{\ln N} - \frac{d}{2 k \ln N}\left(1 + O(k^{-1})\right).
$$
\end{proof}
